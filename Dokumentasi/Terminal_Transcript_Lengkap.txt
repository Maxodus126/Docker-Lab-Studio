D. CRUD MAHASISWA (Gambar 2.4)
PS C:\DockerLabStudio> $nim = "DLT202610202245"
PS C:\DockerLabStudio> $body = @{ name = "Docker Lab Test CRUD"; nim = $nim }
PS C:\DockerLabStudio> $response = Invoke-WebRequest -Uri "http://127.0.0.1:5000/students" -Method Post -Body $body -MaximumRedirection 5 -UseBasicParsing
PS C:\DockerLabStudio> $page = Invoke-WebRequest -Uri "http://127.0.0.1:5000/" -UseBasicParsing
PS C:\DockerLabStudio> $page.Content.Contains($nim)
True
PS C:\DockerLabStudio> # Ambil ID dari baris HTML yang memuat NIM, jangan hardcode ID.
PS C:\DockerLabStudio> $rowPattern = '(?is)<tr\b[^>]*>(?:(?!</tr>).)*?' + [regex]::Escape($nim) + '(?:(?!</tr>).)*?</tr>'
PS C:\DockerLabStudio> $row = [regex]::Match($page.Content, $rowPattern).Value
PS C:\DockerLabStudio> $idMatch = [regex]::Match($row, 'action="/students/(\d+)/delete"')
PS C:\DockerLabStudio> $studentId = [int]$idMatch.Groups[1].Value
PS C:\DockerLabStudio> $deleteUri = "http://127.0.0.1:5000/students/$studentId/delete"
PS C:\DockerLabStudio> $null = Invoke-WebRequest -Uri $deleteUri -Method Post -MaximumRedirection 5 -UseBasicParsing
PS C:\DockerLabStudio> $after = Invoke-WebRequest -Uri "http://127.0.0.1:5000/" -UseBasicParsing
PS C:\DockerLabStudio> $after.Content.Contains($nim)
False
